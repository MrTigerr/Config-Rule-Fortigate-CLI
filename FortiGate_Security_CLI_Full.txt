# PHẦN 1 — AV PROFILE
# Chỉ tạo IS-General: HTTP block, SMTP block, không outbreak-prevention
# ══════════════════════════════════════════════════════

config antivirus profile
    edit "IS-General"
        set comment "Scan files and block viruses."
        config http
            set av-scan block
        end
        config smtp
            set av-scan block
            set executables virus
        end
    next
end
