# ══════════════════════════════════════════════════════
# PHẦN 7 — TỐI ƯU THÊM
# Tắt log-all-domain, đã apply trong DNS profile bên trên
# Nếu cần apply riêng:
# ══════════════════════════════════════════════════════

config dnsfilter profile
    edit "IS-General"
        set log-all-domain disable
    next
    edit "IS-SEO"
        set log-all-domain disable
    next
    edit "IS-TCKT"
        set log-all-domain disable
    next
end
